# SWObjects build rules.
#
# Feature selection that used to live in the CONFIG file is now done with
# cmake cache options, e.g.:
#   cmake -B build -DSWOBJ_XML_PARSER=LIBXML2 -DSWOBJ_SQL_MYSQL=ON
#   cmake --build build -j
#   ctest --test-dir build
cmake_minimum_required(VERSION 3.22)
project(SWObjects VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()

# ---------------------------------------------------------------- options ---
# (successors to the old CONFIG file)

set(SWOBJ_XML_PARSER "EXPAT1" CACHE STRING
    "SAX XML parser library: EXPAT1, LIBXML2 or DISABLED. Needed for query federation and the SPARQL tests.")
set_property(CACHE SWOBJ_XML_PARSER PROPERTY STRINGS EXPAT1 LIBXML2 DISABLED)

option(SWOBJ_UTF8_CONSOLE
       "Print console tables with UTF-8 box-drawing characters instead of ASCII art." ON)

option(SWOBJ_HTTP_CLIENT
       "HTTP client support via Boost.Asio. Needed for query federation." ON)

option(SWOBJ_HTTP_SERVER
       "HTTP server support via Boost.Asio. Needed for the SPARQL server." ON)

option(SWOBJ_SQL_MYSQL    "Build the MySQL SQL client."      OFF)
option(SWOBJ_SQL_POSTGRES "Build the PostgreSQL SQL client." OFF)
option(SWOBJ_SQL_ODBC     "Build the ODBC SQL client."       OFF)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(_swobj_inotify_default ON)
else()
  set(_swobj_inotify_default OFF)
endif()
option(SWOBJ_INOTIFY
       "Watch served files for changes with inotify (Linux only)." ${_swobj_inotify_default})

# --------------------------------------------------------------- toolchain ---

# Homebrew installs bison/flex keg-only; prefer them over the ancient
# /usr/bin/bison that Apple ships.
if(APPLE)
  list(APPEND CMAKE_PREFIX_PATH
       /opt/homebrew/opt/bison /opt/homebrew/opt/flex
       /usr/local/opt/bison /usr/local/opt/flex)
endif()

find_package(BISON 3.2 REQUIRED)
find_package(FLEX 2.6 REQUIRED)

set(_boost_components
    regex thread filesystem date_time program_options iostreams
    log log_setup unit_test_framework)
find_package(Boost 1.67 CONFIG REQUIRED COMPONENTS ${_boost_components})

if(SWOBJ_XML_PARSER STREQUAL "EXPAT1")
  find_package(EXPAT REQUIRED)
elseif(SWOBJ_XML_PARSER STREQUAL "LIBXML2")
  find_package(LibXml2 REQUIRED)
endif()

# version.h used to be generated by the Makefile from `git log`.
find_package(Git QUIET)
set(SWOBJ_GIT_URL "")
set(SWOBJ_GIT_REVISION "")
set(SWOBJ_GIT_AUTHOR "")
set(SWOBJ_GIT_DATE "")
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} config --get remote.origin.url
                  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                  OUTPUT_VARIABLE SWOBJ_GIT_URL OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET)
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-list --count HEAD
                  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                  OUTPUT_VARIABLE SWOBJ_GIT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET)
  execute_process(COMMAND ${GIT_EXECUTABLE} log -1 --format=%an\ <%ae>
                  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                  OUTPUT_VARIABLE SWOBJ_GIT_AUTHOR OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET)
  execute_process(COMMAND ${GIT_EXECUTABLE} log -1 --format=%ci
                  WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                  OUTPUT_VARIABLE SWOBJ_GIT_DATE OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_QUIET)
endif()
configure_file(cmake/version.h.in ${CMAKE_CURRENT_BINARY_DIR}/version.h)

# ----------------------------------------------------------------- parsers ---
# Each parser is a bison grammar (lib/<stem>Parser.ypp) plus a flex scanner
# (lib/<stem>Scanner.lpp). All parsers live in namespace w3c_sw and share the
# w3c_sw::location type from lib/location.hh via ParserCommon.hpp.

set(PARSER_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/parsers)
file(MAKE_DIRECTORY ${PARSER_GEN_DIR})

set(SWOBJ_PARSER_STEMS
    SPARQL SPARQLalgebra Turtle Trig MapSet SQL JSONresults ShExC)

set(SWOBJ_GENERATED_SOURCES "")
foreach(stem IN LISTS SWOBJ_PARSER_STEMS)
  bison_target(${stem}Parser lib/${stem}Parser.ypp
               ${PARSER_GEN_DIR}/${stem}Parser.cpp
               DEFINES_FILE ${PARSER_GEN_DIR}/${stem}Parser.hpp)
  flex_target(${stem}Scanner lib/${stem}Scanner.lpp
              ${PARSER_GEN_DIR}/${stem}Scanner.cpp)
  add_flex_bison_dependency(${stem}Scanner ${stem}Parser)
  list(APPEND SWOBJ_GENERATED_SOURCES
       ${BISON_${stem}Parser_OUTPUTS} ${FLEX_${stem}Scanner_OUTPUTS})
endforeach()

# ----------------------------------------------------------------- library ---

add_library(SWObjects STATIC
  lib/exs.cpp
  lib/ParserCommon.cpp
  lib/RdfDB.cpp
  lib/RdfQueryDB.cpp
  lib/ResultSet.cpp
  lib/ShExSchema.cpp
  lib/SWObjects.cpp
  ${SWOBJ_GENERATED_SOURCES}
)

target_include_directories(SWObjects PUBLIC
  ${CMAKE_CURRENT_BINARY_DIR}   # config.h, version.h
  ${PARSER_GEN_DIR}             # generated parser headers
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/lib
  ${CMAKE_CURRENT_SOURCE_DIR}/interface
  ${CMAKE_CURRENT_SOURCE_DIR}/utf8_v2.3.2_source
  ${FLEX_INCLUDE_DIRS}
)

target_link_libraries(SWObjects PUBLIC
  Boost::regex Boost::thread Boost::filesystem Boost::date_time
  Boost::iostreams Boost::log Boost::log_setup
)

if(SWOBJ_XML_PARSER STREQUAL "EXPAT1")
  target_link_libraries(SWObjects PUBLIC EXPAT::EXPAT)
elseif(SWOBJ_XML_PARSER STREQUAL "LIBXML2")
  target_link_libraries(SWObjects PUBLIC LibXml2::LibXml2)
endif()

if(SWOBJ_SQL_MYSQL)
  find_library(MYSQL_CLIENT_LIB NAMES mysqlclient REQUIRED)
  find_path(MYSQL_INCLUDE_DIR NAMES mysql/mysql.h REQUIRED)
  target_include_directories(SWObjects PUBLIC ${MYSQL_INCLUDE_DIR})
  target_link_libraries(SWObjects PUBLIC ${MYSQL_CLIENT_LIB})
endif()
if(SWOBJ_SQL_POSTGRES)
  find_library(POSTGRES_CLIENT_LIB NAMES pq REQUIRED)
  find_path(POSTGRES_INCLUDE_DIR NAMES libpq-fe.h PATH_SUFFIXES postgresql REQUIRED)
  target_include_directories(SWObjects PUBLIC ${POSTGRES_INCLUDE_DIR})
  target_link_libraries(SWObjects PUBLIC ${POSTGRES_CLIENT_LIB})
endif()
if(SWOBJ_SQL_ODBC)
  find_library(ODBC_CLIENT_LIB NAMES odbc REQUIRED)
  target_link_libraries(SWObjects PUBLIC ${ODBC_CLIENT_LIB})
endif()

# The parsers' debug facilities are compiled in; keep assertions meaningful.
target_compile_options(SWObjects PRIVATE -Wall -Wextra
  -Wno-unused-parameter -Wno-deprecated-declarations)

# ------------------------------------------------------------- executables ---

# The tools land in the source-tree bin/ directory (gitignored), as with the
# old Makefile; the tests shell out to ../bin/sparql .
foreach(tool sparql parseAndFree)
  add_executable(${tool} bin/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE SWObjects
    Boost::program_options Boost::filesystem)
  set_target_properties(${tool} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bin)
endforeach()

if(NOT SQL_CLIENT_NONE)
  add_executable(dm-materialize bin/dm-materialize.cpp)
  target_link_libraries(dm-materialize PRIVATE SWObjects
    Boost::program_options Boost::filesystem)
  set_target_properties(dm-materialize PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bin)
endif()

# A loadable module exercised by test_SPARQL's function_library case
# (dlopen'd via sparql --function-library examples/functionExtension.so).
# It must NOT link its own copy of SWObjects: its references resolve against
# the host binary's symbols at dlopen time, so it shares the host's function
# registry and TTerm singletons.
add_library(functionExtension MODULE examples/functionExtension.cpp)
target_include_directories(functionExtension PRIVATE
  ${CMAKE_CURRENT_BINARY_DIR} ${PARSER_GEN_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/lib
  ${CMAKE_CURRENT_SOURCE_DIR}/interface
  ${CMAKE_CURRENT_SOURCE_DIR}/utf8_v2.3.2_source)
target_include_directories(functionExtension SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
set_target_properties(functionExtension PROPERTIES
  PREFIX ""
  SUFFIX ".so"
  LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/examples)
if(APPLE)
  target_link_options(functionExtension PRIVATE -undefined dynamic_lookup)
endif()
# Export the sparql binary's symbols to dlopen'd modules.
set_target_properties(sparql PROPERTIES ENABLE_EXPORTS ON)

# --------------------------------------------------------------------- tests ---

include(CTest)
if(BUILD_TESTING)
  file(GLOB SWOBJ_TEST_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_*.cpp)
  foreach(src IN LISTS SWOBJ_TEST_SOURCES)
    get_filename_component(name ${src} NAME_WE)
    add_executable(${name} ${src})
    target_link_libraries(${name} PRIVATE SWObjects Boost::unit_test_framework)
    # test_DM needs a SQL client plus manifest/connection arguments
    # (see the old Makefile's t_DM target); build it but don't register it.
    if(name STREQUAL "test_DM")
      continue()
    endif()
    # Tests expect to run from the tests/ directory.
    add_test(NAME ${name}
             COMMAND ${name}
             WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
    # Some tests spawn the sparql binary as a server.
    set_tests_properties(${name} PROPERTIES
      ENVIRONMENT "W3C_SW_SPARQL_BINARY=$<TARGET_FILE:sparql>")
  endforeach()
endif()
